# Real controller/view callback scheduling, with no input/time advance or GPU.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/controller.c" td_ui_retry_owner)
set(td_ui_retry_wire "app.ui_dirty=!td_view_ui(&app.view,&app.rules,app.menu,app.selected,app.notice);")
string(REPLACE "${td_ui_retry_wire}" "" td_ui_without_retry "${td_ui_retry_owner}")
string(LENGTH "${td_ui_retry_owner}" td_ui_before)
string(LENGTH "${td_ui_without_retry}" td_ui_after)
string(LENGTH "${td_ui_retry_wire}" td_ui_wire_length)
math(EXPR td_ui_wire_count "(${td_ui_before}-${td_ui_after})/${td_ui_wire_length}")
if(NOT td_ui_wire_count EQUAL 3)
    message(FATAL_ERROR "TD UI retry control requires all three actual result consumers")
endif()
string(REPLACE "${td_ui_retry_wire}"
    "(void)td_view_ui(&app.view,&app.rules,app.menu,app.selected,app.notice); app.ui_dirty=false;"
    td_ui_old_clear "${td_ui_retry_owner}")
set(td_ui_old_clear_path "${CMAKE_CURRENT_BINARY_DIR}/td_controller_old_clear_dirty.c")
unset(td_ui_previous)
if(EXISTS "${td_ui_old_clear_path}")
    file(READ "${td_ui_old_clear_path}" td_ui_previous)
endif()
if(NOT td_ui_previous STREQUAL td_ui_old_clear)
    file(WRITE "${td_ui_old_clear_path}" "${td_ui_old_clear}")
endif()

# Exercise the production consumer through the installed public runtime and C VM.
# A generated control restores wall resampling only at gameplay consumption.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/controller.c" td_clock_owner)
set(td_clock_wire "else td_tick(&app.rules,dt);")
string(REGEX MATCHALL "else td_tick\\(&app.rules,dt\\)" td_clock_wires "${td_clock_owner}")
list(LENGTH td_clock_wires td_clock_wire_count)
if(NOT td_clock_wire_count EQUAL 1)
    message(FATAL_ERROR "TD clock negative control requires exactly one production dt wire")
endif()
set(td_clock_old_wire [=[else {
                    uint64_t now=jce_time_perf_counter(),freq=jce_time_perf_freq();
                    float elapsed=dt;
                    if(!getenv("JCE_FRAME_DT_FIXED") && app.wall_last && freq && now>=app.wall_last)
                        elapsed=(float)((double)(now-app.wall_last)/(double)freq);
                    app.wall_last=now;
                    td_tick(&app.rules,elapsed);
                }]=])
string(REPLACE "${td_clock_wire}" "${td_clock_old_wire}" td_clock_old_owner "${td_clock_owner}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/td_controller_old_wall.c" "${td_clock_old_owner}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "src/controller.c")
foreach(td_clock_target td_controller_clock_test td_controller_clock_old_wall_test)
    add_executable(${td_clock_target} src/test_controller_clock.c src/rules.c)
    target_include_directories(${td_clock_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_definitions(${td_clock_target} PRIVATE JCE_SCRIPT_MODULE_NO_ENTRY)
    target_link_libraries(${td_clock_target} PRIVATE JCE::JCE)
    jce_script_enable(${td_clock_target} LANGUAGES c REQUIRED NO_STAGE NATIVE_FACTORIES td_module)
    if(MSVC)
        target_compile_options(${td_clock_target} PRIVATE /we4013)
    else()
        target_compile_options(${td_clock_target} PRIVATE -Werror=implicit-function-declaration)
        target_link_libraries(${td_clock_target} PRIVATE m)
    endif()
endforeach()
target_compile_definitions(td_controller_clock_old_wall_test PRIVATE
    TD_CONTROLLER_SOURCE="${CMAKE_CURRENT_BINARY_DIR}/td_controller_old_wall.c")

# Actual runtime/C VM/controller census controls. No Engine ABI or GPU device.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/qa_update_census.inc.h" td_census_owner)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "src/controller.c" "src/qa_update_census.inc.h")
set(td_census_entry_wire "const uint32_t qa_generation=td_qa_update_census_enter();")
set(td_census_return_wire "td_qa_update_census_return(qa_generation);")
set(td_census_update_wire "JCE_C_ON_UPDATE(on_update)")
set(td_census_reset_wire "++td_qa_census.generation;")
set(td_census_overflow_wire "if (*value==UINT32_MAX) td_qa_census.saturated=1;\n    else ++*value;")
foreach(td_census_variant normal missing_update missing_entry missing_return reset overflow)
    set(td_census_controller "${td_clock_owner}")
    set(td_census_header "${td_census_owner}")
    if(NOT td_census_variant STREQUAL "normal")
        if(td_census_variant STREQUAL "missing_update")
            set(td_census_wire "${td_census_update_wire}")
            set(td_census_replacement "JCE_C_ON_UPDATE(NULL)")
        elseif(td_census_variant STREQUAL "missing_entry")
            set(td_census_wire "${td_census_entry_wire}")
            set(td_census_replacement "const uint32_t qa_generation=td_qa_census.generation;")
        elseif(td_census_variant STREQUAL "missing_return")
            set(td_census_wire "${td_census_return_wire}")
            set(td_census_replacement "(void)qa_generation;")
        elseif(td_census_variant STREQUAL "reset")
            set(td_census_wire "${td_census_reset_wire}")
            set(td_census_replacement "td_qa_census.generation=1;")
        else()
            set(td_census_wire "${td_census_overflow_wire}")
            set(td_census_replacement "++*value;")
        endif()
        if(td_census_variant STREQUAL "reset" OR td_census_variant STREQUAL "overflow")
            set(td_census_subject "${td_census_header}")
        else()
            set(td_census_subject "${td_census_controller}")
        endif()
        string(FIND "${td_census_subject}" "${td_census_wire}" td_census_first)
        string(FIND "${td_census_subject}" "${td_census_wire}" td_census_last REVERSE)
        if(td_census_first LESS 0 OR NOT td_census_first EQUAL td_census_last)
            message(FATAL_ERROR "TD census control requires one actual ${td_census_variant} wire")
        endif()
        string(REPLACE "${td_census_wire}" "${td_census_replacement}" td_census_subject "${td_census_subject}")
        if(td_census_variant STREQUAL "reset" OR td_census_variant STREQUAL "overflow")
            set(td_census_header "${td_census_subject}")
        else()
            set(td_census_controller "${td_census_subject}")
        endif()
    endif()
    set(td_census_dir "${CMAKE_CURRENT_BINARY_DIR}/update-census/${td_census_variant}")
    file(MAKE_DIRECTORY "${td_census_dir}")
    file(WRITE "${td_census_dir}/controller.c" "${td_census_controller}")
    file(WRITE "${td_census_dir}/census.inc.h" "${td_census_header}")
    set(td_census_target "td_controller_census_${td_census_variant}_test")
    add_executable(${td_census_target} src/test_controller_clock.c src/rules.c)
    target_include_directories(${td_census_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_definitions(${td_census_target} PRIVATE JCE_SCRIPT_MODULE_NO_ENTRY TD_CENSUS_TEST_ONLY=1
        TD_CONTROLLER_SOURCE="${td_census_dir}/controller.c" TD_QA_CENSUS_OWNER="${td_census_dir}/census.inc.h")
    target_link_libraries(${td_census_target} PRIVATE JCE::JCE)
    jce_script_enable(${td_census_target} LANGUAGES c REQUIRED NO_STAGE NATIVE_FACTORIES td_module)
    if(MSVC)
        target_compile_options(${td_census_target} PRIVATE /we4013)
    else()
        target_compile_options(${td_census_target} PRIVATE -Werror=implicit-function-declaration)
        target_link_libraries(${td_census_target} PRIVATE m)
    endif()
    set(td_census_test "td_controller_census_${td_census_variant}")
    add_test(NAME ${td_census_test} COMMAND ${td_census_target})
    if(COMMAND jce_test_build_dependencies)
        jce_test_build_dependencies(${td_census_target})
    endif()
    if(td_census_variant STREQUAL "normal")
        set_tests_properties(${td_census_test} PROPERTIES TIMEOUT 60
            JCE_TEST_DOMAIN consumer JCE_TEST_KIND unit LABELS "unit;consumer;domain:consumer;kind:unit")
    else()
        set_tests_properties(${td_census_test} PROPERTIES TIMEOUT 60 WILL_FAIL TRUE
            JCE_TEST_DOMAIN consumer JCE_TEST_KIND contract
            LABELS "contract;consumer;negative-control;domain:consumer;kind:contract")
    endif()
    if(DEFINED JCE_TEST_ENVIRONMENT)
        set_tests_properties(${td_census_test} PROPERTIES ENVIRONMENT "${JCE_TEST_ENVIRONMENT}")
        set_property(GLOBAL APPEND PROPERTY JCE_TESTS_WITH_ENV ${td_census_test})
    endif()
endforeach()

endif() # Native-only tests, bundle tool and editor scene module.
