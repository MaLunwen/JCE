cmake_minimum_required(VERSION 3.24)
project(jce_xp_flecs_controls LANGUAGES C)
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(JCE_WINXP_LEGACY ON CACHE BOOL "Exercise the actual XP platform contract" FORCE)
include("${_root}/cmake/JCEWindowsTarget.cmake")
if(NOT IS_ABSOLUTE "${JCE_XP_FLECS_INCLUDE}" OR
   NOT EXISTS "${JCE_XP_FLECS_INCLUDE}/flecs.h")
  message(FATAL_ERROR "Supply the explicit unchanged Flecs include directory")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(xp_flecs_options INTERFACE)
target_include_directories(xp_flecs_options INTERFACE
  "${_root}/engine/include" "${_root}/engine/src" "${JCE_XP_FLECS_INCLUDE}")
target_compile_definitions(xp_flecs_options INTERFACE flecs_STATIC)
add_library(xp_flecs_entropy OBJECT "${_root}/engine/src/os/platform/jce_entropy.c")
target_link_libraries(xp_flecs_entropy PRIVATE xp_flecs_options)
foreach(_kind IN ITEMS bridge modern track atomic)
  if(_kind STREQUAL "bridge")
    set(_target xp_flecs_bridge_control)
  elseif(_kind STREQUAL "modern")
    set(_target xp_flecs_bridge_modern_control)
  elseif(_kind STREQUAL "track")
    set(_target xp_flecs_bridge_track_control)
  else()
    set(_target xp_flecs_atomic_control)
  endif()
  if(_kind STREQUAL "atomic")
    set(_body atomic_control.c)
  else()
    set(_body bridge_control.c)
  endif()
  add_library(${_target}_body OBJECT "${_body}")
  target_link_libraries(${_target}_body PRIVATE xp_flecs_options)
  target_compile_definitions(${_target}_body PRIVATE main=jce_xp_flecs_control_main)
  if(_kind STREQUAL "bridge" OR _kind STREQUAL "track")
    target_compile_definitions(${_target}_body PRIVATE
      FLECS_NO_HTTP=1 FLECS_NO_REST=1 FLECS_NO_OS_API_IMPL=1)
  endif()
  if(_kind STREQUAL "track")
    target_compile_definitions(${_target}_body PRIVATE FLECS_TRACK_OS_ALLOC=1)
  endif()
  add_executable(${_target} main.c $<TARGET_OBJECTS:${_target}_body>
    $<TARGET_OBJECTS:xp_flecs_entropy>)
  target_link_libraries(${_target} PRIVATE xp_flecs_options advapi32)
  target_link_options(${_target} PRIVATE -static -pthread
    -Wl,--major-os-version=5,--minor-os-version=${JCE_XP_PE_MINOR},--major-subsystem-version=5,--minor-subsystem-version=${JCE_XP_PE_MINOR})
  set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
endforeach()

if(JCE_XP_FLECS_MUTANT_DIR)
  file(READ "${JCE_XP_FLECS_MUTANT_DIR}/manifest.json" _manifest)
  foreach(_name IN ITEMS omitted_result missing_join old_time_units wrong_condition wrong_heap_counter missing_set)
    string(JSON _body GET "${_manifest}" mutants "${_name}" path)
    if(NOT IS_ABSOLUTE "${_body}" OR NOT EXISTS "${_body}")
      message(FATAL_ERROR "Required generated first-party mutant is missing")
    endif()
    set(_target "xp_flecs_mutant_${_name}")
    add_library(${_target}_body OBJECT bridge_control.c)
    target_link_libraries(${_target}_body PRIVATE xp_flecs_options)
    target_compile_definitions(${_target}_body PRIVATE main=jce_xp_flecs_control_main
      FLECS_NO_HTTP=1 FLECS_NO_REST=1 FLECS_NO_OS_API_IMPL=1
      "JCE_XP_FLECS_BRIDGE_SOURCE=\"${_body}\"")
    add_executable(${_target} main.c $<TARGET_OBJECTS:${_target}_body>
      $<TARGET_OBJECTS:xp_flecs_entropy>)
    target_link_libraries(${_target} PRIVATE xp_flecs_options advapi32)
    target_link_options(${_target} PRIVATE -static -pthread
      -Wl,--major-os-version=5,--minor-os-version=${JCE_XP_PE_MINOR},--major-subsystem-version=5,--minor-subsystem-version=${JCE_XP_PE_MINOR})
    set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
  endforeach()
endif()
