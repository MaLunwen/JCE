cmake_minimum_required(VERSION 3.24)
project(jce_xp_os_controls LANGUAGES C)
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(JCE_WINXP_LEGACY ON CACHE BOOL "Exercise actual XP platform admission" FORCE)
include("${_root}/cmake/JCEWindowsTarget.cmake")
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(_owner "${CMAKE_CURRENT_SOURCE_DIR}|${CMAKE_C_COMPILER}|${JCE_XP_TOOLCHAIN_ROOT}|${JCE_XP_TRIPLE}")
if(DEFINED JCE_XP_OS_CACHE_OWNER AND NOT JCE_XP_OS_CACHE_OWNER STREQUAL _owner)
  message(FATAL_ERROR "XP OS control cache belongs to a different source/compiler/toolchain")
endif()
set(JCE_XP_OS_CACHE_OWNER "${_owner}" CACHE INTERNAL "XP OS control owner")
set(_header "${_root}/engine/src/os/platform/jce_executable_path_native_internal.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_header}")
file(READ "${_header}" _source)
set(_wire "if (!jce_fs_executable_path_utf16_valid_internal(wide, size)) {")
string(FIND "${_source}" "${_wire}" _found)
if(_found LESS 0)
  message(FATAL_ERROR "XP executable-path validation wire changed")
endif()
string(REPLACE "${_wire}" "if (false) {" _omitted "${_source}")
include("${_root}/tests/cmake/JceGeneratedFixture.cmake")
jce_test_write_if_changed("${CMAKE_BINARY_DIR}/omitted_utf16.h" "${_omitted}")
add_library(xp_os_entropy OBJECT "${_root}/engine/src/os/platform/jce_entropy.c")
target_include_directories(xp_os_entropy PRIVATE "${_root}/engine/include")
add_executable(jce_xp_executable_path executable_path.c)
add_executable(jce_xp_executable_path_omitted executable_path.c)
target_compile_definitions(jce_xp_executable_path_omitted PRIVATE
  "JCE_EXECUTABLE_PATH_HEADER=\"${CMAKE_BINARY_DIR}/omitted_utf16.h\"")
add_executable(jce_xp_window_adapter window_adapter.c
  "${_root}/engine/src/os/platform/jce_window_adapter.c")
foreach(_target IN ITEMS jce_xp_executable_path jce_xp_executable_path_omitted jce_xp_window_adapter)
  target_include_directories(${_target} PRIVATE "${_root}/engine/include" "${_root}/engine/src")
  target_sources(${_target} PRIVATE $<TARGET_OBJECTS:xp_os_entropy>)
  target_link_libraries(${_target} PRIVATE advapi32)
  target_link_options(${_target} PRIVATE -static
    -Wl,--major-os-version=5,--minor-os-version=${JCE_XP_PE_MINOR},--major-subsystem-version=5,--minor-subsystem-version=${JCE_XP_PE_MINOR})
  set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
endforeach()
