cmake_minimum_required(VERSION 3.24)
project(jce_xp_embedding_probe LANGUAGES C CXX RC)
get_filename_component(_jce_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(JCE_WINXP_LEGACY ON CACHE BOOL "Use the actual XP target contract" FORCE)
include("${_jce_root}/cmake/JCEWindowsTarget.cmake")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Windows" OR
   NOT DEFINED JCE_XP_PE_MINOR OR NOT DEFINED JCE_XP_TOOLCHAIN_ROOT)
  message(FATAL_ERROR "Use the existing JCE XP toolchain with an explicit root")
endif()
if(NOT JCE_XP_EMBED_SUBSYSTEMS STREQUAL "console3-gui2")
  message(FATAL_ERROR "Embedding cache must declare both console and GUI contracts")
endif()
if(JCE_XP_EXPECTED_ARCH STREQUAL "x86" AND CMAKE_SIZEOF_VOID_P EQUAL 4)
elseif(JCE_XP_EXPECTED_ARCH STREQUAL "x64" AND CMAKE_SIZEOF_VOID_P EQUAL 8)
else()
  message(FATAL_ERROR "Requested embedding architecture differs from the actual target")
endif()
file(TO_CMAKE_PATH "${JCE_XP_TOOLCHAIN_ROOT}/bin/${JCE_XP_TRIPLE}-windres.exe" _expected_rc)
file(TO_CMAKE_PATH "${CMAKE_RC_COMPILER}" _actual_rc)
if(NOT _actual_rc STREQUAL _expected_rc OR NOT EXISTS "${_expected_rc}")
  message(FATAL_ERROR "Use the exact target-prefixed XP windres executable")
endif()
if(NOT IS_ABSOLUTE "${JCE_XP_EMBED_PAYLOAD}" OR NOT EXISTS "${JCE_XP_EMBED_PAYLOAD}")
  message(FATAL_ERROR "An absolute in-build embedding payload is required")
endif()
file(READ "${JCE_XP_EMBED_PAYLOAD}" _payload_hex HEX)
if(NOT _payload_hex STREQUAL "317ca5e21963bd4f")
  message(FATAL_ERROR "Embedding payload does not match the eight-byte fixture")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(xp_probe_options INTERFACE)
target_include_directories(xp_probe_options INTERFACE "${_jce_root}/engine/include")
target_compile_definitions(xp_probe_options INTERFACE WIN32_LEAN_AND_MEAN=1)
target_compile_options(xp_probe_options INTERFACE "$<$<COMPILE_LANGUAGE:C,CXX>:-O2;-pthread>")
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_compile_options(xp_probe_options INTERFACE $<$<COMPILE_LANGUAGE:C,CXX>:-msse2>)
endif()
add_library(xp_core_probe OBJECT "${_jce_root}/tests/compatibility/xp_core/main.cpp")
target_compile_definitions(xp_core_probe PRIVATE main=jce_xp_core_main)
target_link_libraries(xp_core_probe PRIVATE xp_probe_options)
file(TO_CMAKE_PATH "${JCE_XP_EMBED_PAYLOAD}" JCE_XP_EMBED_PAYLOAD_RC)
configure_file("${CMAKE_CURRENT_LIST_DIR}/resource.rc.in" "${CMAKE_BINARY_DIR}/probe.rc" @ONLY)
add_library(xp_embedding_body OBJECT "${CMAKE_CURRENT_LIST_DIR}/main.cpp"
  "${_jce_root}/engine/src/os/platform/jce_entropy.c")
target_link_libraries(xp_embedding_body PRIVATE xp_probe_options)
# Both subsystem targets consume the one production SDK embedding helper.
include("${_jce_root}/cmake/JCESDKHelpers.cmake")
add_library(xp_embedding_assets OBJECT)
target_link_libraries(xp_embedding_assets PRIVATE xp_probe_options)
_jce_target_embed_binary(xp_embedding_assets "${JCE_XP_EMBED_PAYLOAD}"
  assets_pak_data "${CMAKE_BINARY_DIR}/payload.S")
foreach(_target IN ITEMS jce_xp_embedding_probe jce_xp_embedding_gui_probe)
  if(_target STREQUAL "jce_xp_embedding_gui_probe")
    add_executable(${_target} WIN32 "${CMAKE_BINARY_DIR}/probe.rc")
  else()
    add_executable(${_target} "${CMAKE_BINARY_DIR}/probe.rc")
  endif()
  target_sources(${_target} PRIVATE $<TARGET_OBJECTS:xp_core_probe>
    $<TARGET_OBJECTS:xp_embedding_body> $<TARGET_OBJECTS:xp_embedding_assets>)
  target_link_libraries(${_target} PRIVATE xp_probe_options advapi32)
  target_link_options(${_target} PRIVATE -static -pthread
    -Wl,--major-os-version=5,--minor-os-version=${JCE_XP_PE_MINOR},--major-subsystem-version=5,--minor-subsystem-version=${JCE_XP_PE_MINOR})
  set_target_properties(${_target} PROPERTIES LINKER_LANGUAGE CXX
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
endforeach()
