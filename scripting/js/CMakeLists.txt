cmake_minimum_required(VERSION 3.20)

# ------------------------------------------------------------------ #
#  jce_script_vm_js — the "js" JceScriptVM over quickjs-ng.           #
#                                                                     #
#  FROM CONAN, like lua/5.4.8 and for the same reason: an embedded    #
#  language runtime is a dependency, and two of them acquired two     #
#  different ways is a difference every later reader has to learn.    #
#                                                                     #
#  The recipe is LOCAL (conan/recipes/quickjs-ng) because             #
#  conan-center ships Bellard's fork, whose recipe raises             #
#  ConanInvalidConfiguration on msvc -- an honest refusal, since that #
#  tree is Makefile-only with no MSVC support, so no conan hook could #
#  fix it.  quickjs-ng ships CMake and builds under MSVC unchanged.   #
#  scripts/jce.py exports conan/recipes/*/ before every install.      #
#                                                                     #
#  NO TOOLCHAIN PROBE, and that is the rule not applying rather than  #
#  an exception to it: python probes for an embeddable CPython and    #
#  java for a JDK because those runtimes belong to the MACHINE.  This #
#  one is a conan dependency like any other -- if it resolved, it     #
#  works.  What CAN be absent is the package itself (an engine build  #
#  configured before this dependency existed), and that is what the   #
#  guard below reports, by name.                                      #
# ------------------------------------------------------------------ #

find_package(quickjs-ng QUIET)
if(NOT TARGET quickjs-ng::quickjs-ng)
	message(STATUS
		"JCE scripting: JS backend SKIPPED — quickjs-ng did not resolve. Its "
		"recipe is local (conan/recipes/quickjs-ng) and is exported by "
		"scripts/jce.py before every conan install; a build directory "
		"configured before that existed will not have it. Re-run the conan "
		"install for this build dir.")
	return()
endif()

add_library(jce_script_vm_js STATIC
	"${CMAKE_CURRENT_SOURCE_DIR}/src/jce_script_vm_js.c"
	# GENERATED by tools/scriptgen/emit_js.py and COMMITTED.  Listed by name
	# rather than globbed: a glob would silently build a stale copy that a
	# regenerate had already replaced elsewhere, and would build nothing at
	# all -- with no error -- if the file were deleted.
	"${CMAKE_CURRENT_SOURCE_DIR}/src/jce_script_bindings_js.gen.c")
set_target_properties(jce_script_vm_js PROPERTIES
	C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
target_include_directories(jce_script_vm_js PUBLIC
	"${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(jce_script_vm_js PUBLIC quickjs-ng::quickjs-ng)
# jce_script and jce_platform, which this archive calls, live in the engine.
target_link_libraries(jce_script_vm_js PRIVATE jce_core)
jce_target_warnings(jce_script_vm_js STRICT)

# The engine archive's PATH, published for the SDK install rules.
#
# Not `if(TARGET quickjs-ng::quickjs-ng)` over in JCEScriptingInstall.cmake:
# find_package creates a DIRECTORY-SCOPED imported target, so that test is
# false one scope up and the install silently did nothing -- measured, the SDK
# shipped jce_script_vm_js.lib and not the quickjs it links.  IMPORTED_GLOBAL
# would fix it from 3.11 and find_package(GLOBAL) from 3.24; a path in a cache
# variable needs neither and is what the install rule actually wants.
# conan's CMakeDeps makes quickjs-ng::quickjs-ng an INTERFACE target and finds
# the archive into a target of its own, so IMPORTED_LOCATION on the public name
# is empty -- measured: reading it produced nothing and the SDK shipped
# jce_script_vm_js.lib without the quickjs it links.  The package's own
# LIB_DIRS variables are what the generated data file publishes, so look there.
find_library(JCE_QUICKJS_LIBRARY
	NAMES qjs libqjs
	HINTS ${quickjs-ng_LIB_DIRS_RELEASE} ${quickjs-ng_LIB_DIRS_DEBUG}
	      ${quickjs-ng_LIB_DIRS} "${quickjs-ng_PACKAGE_FOLDER_RELEASE}/lib"
	      "${quickjs-ng_PACKAGE_FOLDER_DEBUG}/lib"
	NO_DEFAULT_PATH)
mark_as_advanced(JCE_QUICKJS_LIBRARY)

message(STATUS "JCE scripting: JS backend ON (quickjs-ng, .jcejs)")
